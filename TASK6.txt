    // CHECK PACKAGE EXPIRY BASED ON EXACT DURATION (DAYS)
    function checkPackageExpiration() {
      if (!state.packageId) return;
      if (state.packageExpiresAt) {
        const now = Date.now();
        if (now > state.packageExpiresAt) {
          const expiredPkg = PACKAGES.find(p => p.id === state.packageId);
          state.packageId = null;
          state.packagePurchasedAt = null;
          state.packageExpiresAt = null;
          saveState();
          if (db && state.user) {
            db.ref('users/' + state.user.phone).update({
              package: 'ফ্রি প্যাকেজ',
              packageId: null,
              packageExpiresAt: null
            });
          }
          renderApp();
          showToast(`আপনার ${expiredPkg ? expiredPkg.nameBn : 'প্যাকেজের'} মেয়াদ সমাপ্ত হয়েছে। দয়া করে নতুন প্যাকেজ চালু করুন।`, 'info');
        }
      }
    }

    function checkFreeTasksReset() {
      const today = new Date().toISOString().split('T')[0];
      if (!state.lastFreeResetDate) {
        state.lastFreeResetDate = today;
        saveState();
        return false;
      }
      if (state.lastFreeResetDate !== today) {
        // Daily reset: reset free tasks AND daily VIP tasks so user gets their exact daily quota again
        state.completedTasks = [];
        state.lastFreeResetDate = today;
        checkPackageExpiration();
        saveState();
        return true;
      }
      checkPackageExpiration();
      return false;
    }

    function initAdminGatewayListeners() {
      if (!db) return;

      const handleSettingsUpdate = snap => {
        if (!snap.exists()) return;
        const val = snap.val();
        if (val) {
          const b = val.bkashNumber || val.bkash || val.bKash || val.bkash_number;
          const n = val.nagadNumber || val.nagad || val.Nagad || val.nagad_number;
          if (b) appSettings.bkashNumber = String(b).trim();
          if (n) appSettings.nagadNumber = String(n).trim();

          const targetNum = document.getElementById('targetNumber');
          if (targetNum) {
            targetNum.textContent = currentDepositMethod === 'bKash' ? appSettings.bkashNumber : appSettings.nagadNumber;
          }
        }
      };

      db.ref('settings').on('value', handleSettingsUpdate);
      db.ref('gateways').on('value', handleSettingsUpdate);
      db.ref('admin/settings').on('value', handleSettingsUpdate);
    }

    function attachFirebaseListener(phone) {
      if (!db) return;
      db.ref('users/' + phone).on('value', snapshot => {
        if (snapshot.exists()) {
          const val = snapshot.val();

          // 1. LIVE BAN CHECK
          const isBanned = val.isBanned === true || val.banned === true || val.status === 'banned' || val.status === 'blocked' || val.isBlocked === true;
          if (isBanned) {
            state.isBanned = true;
            saveState();
            showBannedScreen(val.banReason || 'অ্যাডমিন কর্তৃক আপনার অ্যাকাউন্ট স্থগিত (Ban) করা হয়েছে।');
            return;
          } else {
            state.isBanned = false;
            hideBannedScreen();
          }

          // 2. LIVE BALANCE EDIT
          if (val.balance !== undefined && val.balance !== null) {
            const parsedBal = Number(val.balance);
            if (!isNaN(parsedBal) && parsedBal !== state.balance) {
              state.balance = parsedBal;
              saveState();
              renderApp();
            }
          }

          // 3. LIVE PACKAGE & EXPIRATION
          if (val.packageExpiresAt !== undefined) {
            state.packageExpiresAt = val.packageExpiresAt;
          }
          if (val.packageId !== undefined && val.packageId !== state.packageId) {
            state.packageId = val.packageId;
            saveState();
            renderApp();
          } else if (val.package && !state.packageId) {
            const foundPkg = PACKAGES.find(p => p.name.toLowerCase() === String(val.package).toLowerCase() || p.nameBn === val.package);
            if (foundPkg) {
              state.packageId = foundPkg.id;
              saveState();
              renderApp();
            }
          }

          // 4. LIVE KYC
          if (val.isKycVerified !== undefined && val.isKycVerified !== state.isKycVerified) {
            state.isKycVerified = !!val.isKycVerified;
            saveState();
            renderApp();
          }
        }
      });
    }

    function handleSignup(e) {
      e.preventDefault();
      const name = document.getElementById('suName').value.trim();
      const phone = document.getElementById('suPhone').value.trim();
      const pass = document.getElementById('suPass').value.trim();

      if (!phone.startsWith('01') || phone.length !== 11) {
        showToast('সঠিক ১১ সংখ্যার বাংলাদেশি মোবাইল নম্বর দিন', 'error');
        return;
      }
      if (pass.length < 4) {
        showToast('কমপক্ষে ৪ সংখ্যার পিন বা পাসওয়ার্ড দিন', 'error');
        return;
      }

      const btn = document.getElementById('suSubmitBtn');
      btn.disabled = true;
      btn.classList.add('opacity-70');

      if (db) {
        db.ref('users/' + phone).once('value').then(snapshot => {
          if (snapshot.exists()) {
            showToast('এই নম্বরে ইতিমধ্যে অ্যাকাউন্ট আছে! দয়া করে লগইন করুন।', 'error');
            setAuthMode('login');
            document.getElementById('liPhone').value = phone;
            btn.disabled = false;
            btn.classList.remove('opacity-70');
            return;
          }
          registerUserAccount(name, phone, pass);
        }).catch(() => {
          registerUserAccount(name, phone, pass);
        });
      } else {
        registerUserAccount(name, phone, pass);
      }
    }

    function handleLogin(e) {
      e.preventDefault();
      const phone = document.getElementById('liPhone').value.trim();
      const pass = document.getElementById('liPass').value.trim();

      if (!phone.startsWith('01') || phone.length !== 11) {
        showToast('সঠিক ১১ সংখ্যার মোবাইল নম্বর দিন', 'error');
        return;
      }
      if (!pass) {
        showToast('আপনার পাসওয়ার্ড বা গোপন পিন লিখুন', 'error');
        return;
      }

      const btn = document.getElementById('liSubmitBtn');
      btn.disabled = true;
      btn.classList.add('opacity-70');

      if (!db) {
        showToast('ফায়ারবেস ডাটাবেস প্রস্তুত নয়, ইন্টারনেট চেক করুন', 'error');
        btn.disabled = false;
        btn.classList.remove('opacity-70');
        return;
      }

      db.ref('users/' + phone).once('value').then(snapshot => {
        btn.disabled = false;
        btn.classList.remove('opacity-70');

        if (!snapshot.exists()) {
          showToast('এই নম্বরে কোনো অ্যাকাউন্ট পাওয়া যায়নি! আগে রেজিস্ট্রেশন করুন।', 'error');
          return;
        }

        const data = snapshot.val();

        const isBanned = data.isBanned === true || data.banned === true || data.status === 'banned' || data.status === 'blocked';
        if (isBanned) {
          showToast('আপনার অ্যাকাউন্ট স্থগিত (Banned) করা হয়েছে!', 'error');
          showBannedScreen(data.banReason || 'অ্যাডমিন কর্তৃক আপনার অ্যাকাউন্ট স্থগিত (Ban) করা হয়েছে।');
          return;
        }

        if (data.password && data.password !== pass) {
          showToast('ভুল পাসওয়ার্ড বা গোপন পিন! সঠিক পিন দিন।', 'error');
          return;
        }

        state.user = {
          name: data.name || 'ব্যবহারকারী',
          phone: data.phone || phone,
          avatar: data.avatar || 'https://images.unsplash.com/photo-1535713875002-d1d0cf377fde?w=150'
        };
        state.balance = typeof data.balance === 'number' ? data.balance : Number(data.balance) || 50;
        state.packageId = data.packageId || null;
        state.packageExpiresAt = data.packageExpiresAt || null;
        state.isKycVerified = !!data.isKycVerified;
        state.isBanned = false;

        saveState();
        checkPackageExpiration();
        attachFirebaseListener(phone);
        renderApp();
        showToast(`লগইন সফল! স্বাগতম, ${state.user.name.split(' ')[0]}`, 'success');
      }).catch(err => {
        btn.disabled = false;
        btn.classList.remove('opacity-70');
        showToast('লগইন করতে সমস্যা হচ্ছে। ইন্টারনেট চেক করুন।', 'error');
      });
    }

    function initiateTask(taskId, isFree) {
      if (isFree) {
        checkFreeTasksReset();
        const doneCount = FREE_TASKS.filter(t => state.completedTasks.includes(t.id)).length;
        if (doneCount >= 2 && !state.completedTasks.includes(taskId)) {
          showToast('আজকের ফ্রি কাজের কোটা (২/২) সম্পন্ন হয়েছে। পরবর্তী ২৪ ঘণ্টার জন্য অপেক্ষা করুন।', 'error');
          return;
        }
      } else if (!state.packageId) {
        showToast('ভিআইপি কাজের জন্য যেকোনো প্যাকেজ সক্রিয় করুন', 'error');
        switchTab('packages');
        return;
      }

      checkPackageExpiration();
      if (!isFree && !state.packageId) {
        showToast('আপনার প্যাকেজের মেয়াদ শেষ হয়েছে। দয়া করে নতুন প্যাকেজ চালু করুন।', 'error');
        return;
      }

      const currentVipTasks = getVipTasksForPackage(state.packageId);
      const all = [...FREE_TASKS, ...currentVipTasks];
      const task = all.find(t => t.id === taskId);
      if (!task) return;

      if (state.completedTasks.includes(task.id)) {
        showToast('এই কাজটি আপনি ইতোমধ্যে সম্পন্ন করেছেন!', 'info');
        return;
      }

      activeTaskToExecute = task;
      document.getElementById('taskModalTitle').textContent = task.title;
      document.getElementById('taskModalReward').textContent = `+৳${task.reward}.00`;
      document.getElementById('taskModalInstructions').textContent = task.desc;
      document.getElementById('taskAnswerInput').value = '';
      openModal('taskModal');
    }

    // AUTO-SELECT PACKAGE AMOUNT ON BUY & CONFIGURE DURATION AND DAILY TASKS
    function buyPackage(pkgId) {
      const pkg = PACKAGES.find(p => p.id === pkgId);
      if (!pkg) return;

      if (state.balance < pkg.price) {
        showToast(`প্যাকেজের মূল্য ৳${pkg.price} টাকা। পেমেন্ট সম্পন্ন করুন।`, 'info');
        openDepositModal(pkg.price);
        return;
      }

      state.balance -= pkg.price;
      state.packageId = pkg.id;
      state.packagePurchasedAt = Date.now();
      state.packageExpiresAt = Date.now() + (pkg.days * 24 * 60 * 60 * 1000);
      
      // Reset completed VIP tasks to immediately unlock the full set of package tasks
      state.completedTasks = (state.completedTasks || []).filter(id => id.startsWith('free-'));

      state.transactions.unshift({
        id: 'tx-' + Date.now(),
        desc: `প্যাকেজ ক্রয়: ${pkg.nameBn}`,

    function calculateWithdrawFee() {
      const amt = Number(document.getElementById('wAmount').value) || 0;
      const fee = (amt * 0.0185);
      const net = Math.max(0, amt - fee);
      document.getElementById('wFee').textContent = `-৳${fee.toFixed(2)}`;
      document.getElementById('wNet').textContent = `৳${net.toFixed(2)}`;
    }

    function submitWithdrawal() {
      const amt = Number(document.getElementById('wAmount').value) || 0;
      const phone = document.getElementById('wPhone').value.trim();

      if (amt < 200) {
        showToast('সর্বনিম্ন উত্তোলনের পরিমাণ ৳২০০ টাকা।', 'error');
        return;
      }
      if (amt > state.balance) {
        showToast(`অপর্যাপ্ত ব্যালেন্স! আপনার বর্তমান ব্যালেন্স ৳${state.balance.toFixed(2)}`, 'error');
        return;
      }
      if (!phone || phone.length < 11) {
        showToast('সঠিক ১১ ডিজিটের পার্সোনাল মোবাইল নম্বর লিখুন', 'error');
        return;
      }

      state.balance -= amt;
      state.transactions.unshift({
        id: 'tx-' + Date.now(),
        desc: `${currentWithdrawMethod} পেমেন্ট (${phone})`,
        amount: -amt,
        date: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })
      });

      if (db && state.user) {
        db.ref('users/' + state.user.phone + '/balance').set(state.balance);
        db.ref('withdrawals').push({
          userName: state.user.name,
          userPhone: phone,
          method: currentWithdrawMethod,
          amount: amt,
          status: 'pending',
          timestamp: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })
        });
      }

      saveState();
      closeModal('withdrawModal');
      renderApp();
      showToast('পেমেন্ট অনুরোধ গৃহীত হয়েছে! ১০-১৫ মিনিটে টাকা পৌঁছাবে।', 'success');
    }

    function startCountdownTimer() {
      function tick() {
        const now = new Date();
        const tomorrow = new Date(now.getFullYear(), now.getMonth(), now.getDate() + 1, 0, 0, 0);
        const diff = tomorrow - now;

        if (diff <= 1000) {
          if (checkFreeTasksReset()) {
            renderFreeTasks();
            renderVipTasks();
            showToast('নতুন দিনের টাস্ক আনলক হয়েছে!', 'info');
          }
        }

        const totalSec = Math.max(0, Math.floor(diff / 1000));
        const h = String(Math.floor(totalSec / 3600)).padStart(2, '0');
        const m = String(Math.floor((totalSec % 3600) / 60)).padStart(2, '0');
        const s = String(totalSec % 60).padStart(2, '0');
        const el = document.getElementById('countdownTimer');
        if (el) el.textContent = `${h}:${m}:${s}`;
      }
      tick();
      setInterval(tick, 1000);
    }
